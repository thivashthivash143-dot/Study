            <div class="bg-white rounded-xl shadow-sm p-6">
                <div class="flex justify-between items-center mb-6">
                    <h2 class="text-xl font-bold text-gray-800">Chapter List</h2>
                    <span id="selected-subject" class="bg-blue-100 text-blue-700 px-3 py-1 rounded-full text-sm font-semibold">Select a Subject</span>
                </div>

                <div id="chapters-container" class="space-y-4">
                    <div class="text-center py-10 text-gray-500">
                        <i class="fa-solid fa-book-open text-4xl mb-3 block opacity-20"></i>
                        Select a subject from the left to see chapters.
                    </div>
                </div>
            </div>
        </main>
    </div>

    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/10.13.0/firebase-app.js";
        import { getFirestore, collection, getDocs } from "https://www.gstatic.com/firebasejs/10.13.0/firebase-firestore.js";

        const firebaseConfig = {
            apiKey: "AIzaSyAn7E7olxRpT_dBvQCL2K5nx-slpuTb1e8",
            authDomain: "study-327d0.firebaseapp.com",
            projectId: "study-327d0",
            storageBucket: "study-327d0.firebasestorage.app",
            messagingSenderId: "718476771682",
            appId: "1:718476771682:web:154c4253a4a68cc2aeec3b",
            measurementId: "G-26V9GF5XP7"
        };

        const app = initializeApp(firebaseConfig);
        const db = getFirestore(app);

        // 1. Get Class ID from URL
        const urlParams = new URLSearchParams(window.location.search);
        const classId = urlParams.get('id'); // e.g., '11th'
        document.getElementById('class-title').innerText = classId + " Standard";

        // 2. Fetch Subjects from Firestore
        async function loadSubjects() {
            const subjectsList = document.getElementById('subjects-list');
            try {
                // Path: classes / 11th / subjects
                const querySnapshot = await getDocs(collection(db, `classes/${classId}/subjects`));
                subjectsList.innerHTML = "";
                
                if (querySnapshot.empty) {
                    subjectsList.innerHTML = "<p class='text-red-400 text-sm'>No subjects found. Add them via Admin Panel.</p>";
                    return;
                }

                querySnapshot.forEach((doc) => {
                    const data = doc.data();
                    const li = document.createElement('li');
                    li.className = "flex items-center justify-between p-3 rounded-lg hover:bg-blue-50 cursor-pointer transition text-gray-700 font-medium border border-transparent hover:border-blue-200";
                    li.innerHTML = `<span>${data.subjectName}</span> <i class="fa-solid fa-chevron-right text-xs text-gray-400"></i>`;
                    li.onclick = () => loadChapters(doc.id, data.subjectName);
                    subjectsList.appendChild(li);
                });
            } catch (e) {
                console.error("Error: ", e);
            }
        }

        // 3. Fetch Chapters based on selected subject
        async function loadChapters(subjectId, subjectName) {
            document.getElementById('selected-subject').innerText = subjectName;
            const container = document.getElementById('chapters-container');
            container.innerHTML = "<p class='text-center py-10'>Loading chapters...</p>";

            try {
                // Path: classes / 11th / subjects / physics / content
                const querySnapshot = await getDocs(collection(db, `classes/${classId}/subjects/${subjectId}/content`));
                container.innerHTML = "";

                if (querySnapshot.empty) {
                    container.innerHTML = "<div class='text-center py-10 text-gray-400'>No chapters added yet for this subject.</div>";
                    return;
                }

                querySnapshot.forEach((doc) => {
                    const data = doc.data();
                    const div = document.createElement('div');
                    div.className = "flex items-center justify-between p-4 border rounded-xl hover:shadow-md transition bg-gray-50";
                    div.innerHTML = `
                        <div class="flex items-center space-x-4">
                            <div class="bg-white p-3 rounded-lg shadow-sm text-blue-600"><i class="fa-solid fa-file-lines"></i></div>
                            <div>
                                <h4 class="font-bold text-gray-800">${data.title}</h4>
                                <p class="text-xs text-gray-500">${data.type || 'Study Material'}</p>
                            </div>
                        </div>
                        <button onclick="window.open('${data.link}', '_blank')" class="bg-blue-600 text-white px-4 py-2 rounded-lg text-sm font-bold hover:bg-blue-700">View</button>
                    `;
                    container.appendChild(div);
                });
            } catch (e) {
                console.error("Error: ", e);
            }
        }

        loadSubjects();
    </script>
