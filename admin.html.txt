    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/10.13.0/firebase-app.js";
        import { getFirestore, doc, setDoc, addDoc, collection } from "https://www.gstatic.com/firebasejs/10.13.0/firebase-firestore.js";

        const firebaseConfig = {
            apiKey: "AIzaSyAn7E7olxRpT_dBvQCL2K5nx-slpuTb1e8",
            authDomain: "study-327d0.firebaseapp.com",
            projectId: "study-327d0",
            storageBucket: "study-327d0.firebasestorage.app",
            messagingSenderId: "718476771682",
            appId: "1:718476771682:web:154c4253a4a68cc2aeec3b",
            measurementId: "G-26V9GF5XP7"
        };

        const app = initializeApp(firebaseConfig);
        const db = getFirestore(app);

        const btn = document.getElementById('submitBtn');

        btn.onclick = async () => {
            const classId = document.getElementById('classSelect').value;
            const subName = document.getElementById('subjectName').value.toLowerCase().trim();
            const title = document.getElementById('contentTitle').value;
            const type = document.getElementById('contentType').value;
            const link = document.getElementById('contentLink').value;

            if(!subName || !title || !link) {
                alert("Ellam fill pannunga boss!");
                return;
            }

            btn.disabled = true;
            btn.innerText = "Adding...";

            try {
                // 1. First, make sure the subject document exists
                const subjectRef = doc(db, `classes/${classId}/subjects/${subName}`);
                await setDoc(subjectRef, {
                    subjectName: subName.toUpperCase()
                }, { merge: true });

                // 2. Add the content inside that subject
                await addDoc(collection(db, `classes/${classId}/subjects/${subName}/content`), {
                    title: title,
                    type: type,
                    link: link,
                    timestamp: new Date()
                });

                alert("Successfully added to " + classId + " " + subName);
                
                // Clear inputs
                document.getElementById('contentTitle').value = "";
                document.getElementById('contentLink').value = "";

            } catch (e) {
                console.error("Error: ", e);
                alert("Error: " + e.message);
            } finally {
                btn.disabled = false;
                btn.innerText = "ADD DATA TO WEBSITE";
            }
        };
    </script>
